def load_portfolio(path: Path, assets: pd.DataFrame | None = None) -> dict:
    """Load portfolio.json, validate weights (sum 100 +- 0.01, known tickers), add `weights` as fractions."""
    pf = json.loads(Path(path).read_text())
    w = pf["weights_pct"]
    total = sum(w.values())
    if abs(total - 100) > 0.01:
        raise ValueError(f"{path}: weights sum to {total}, expected 100")
    if assets is not None:
        unknown = set(w) - set(assets.index)
        if unknown:
            raise ValueError(f"{path}: tickers not in universe: {sorted(unknown)}")
    pf["weights"] = {t: v / 100 for t, v in w.items()}
    return pf

def _intent(old_field: dict | None, out: dict, context: dict) -> tuple[dict | None, GateStatus]:
    """A draft is a buy (intent.apply has no sell semantics, §3.6): any other action is rejected, never confirmable.
    Also: universe ticker, weight in (0, 100], a known funding kind with its parameter."""
    tickers = _universe_tickers(context)
    f = out.get("funding")
    weight = out.get("target_weight_pct")
    try:
        ok = (out.get("action") == "buy" and out.get("ticker") in tickers
              and _is_num(weight) and 0 < weight <= 100
              and isinstance(f, dict) and f.get("kind") in N4_FUNDING_KINDS
              and (f["kind"] != "specific" or f.get("ticker") in tickers)
              and (f["kind"] != "pro_rata_class" or bool(f.get("asset_class")))
              and (f["kind"] != "cash_contribution" or (_is_num(f.get("amount_cad")) and f["amount_cad"] > 0)))
    except (TypeError, KeyError):
        ok = False
    if not ok:
        return _kept(old_field, "rejected:schema")
    return copy.deepcopy(out), "accepted"

def b2_confirm(state: dict, auto_confirm: bool = False) -> dict:
    """B2: the user confirms or edits the draft. The confirmed intent is user-declared data (the fixture intent);
    the draft only prefills the form. Without `auto_confirm` the run stops here (`awaiting_confirmation`).
    `intent_edits` compare the confirmed intent with the blanked draft (D10 pt 1): a field N4 listed as ambiguous
    that the user filled in is a user edit. The draft kept for the audit is not changed."""
    if not auto_confirm:
        return {**state, "awaiting_confirmation": True}
    intent = {**state["fixture_intent"], "status": "confirmed"}
    draft = stated_draft(state.get("draft"))
    edits = [k for k in INTENT_KEYS if draft.get(k) != intent.get(k)]
    return {**state, "awaiting_confirmation": False, "intent": intent, "intent_edits": edits}

def notes_changed(profile: dict) -> bool:
    """True when `values.notes` needs N3: no parsed result yet, or the parsed rules do not account for the notes.
    The notes are unchanged if the stored `input_hash` is the hash of the N3 payload, or (hand-written fixtures,
    hashed differently) if the quotes of the parsed rules cover all of the notes."""
    values = profile.get("values", {})
    notes = values.get("notes")
    if not isinstance(notes, str) or not notes.strip():
        return False
    node = values.get("parsed_notes")
    if not isinstance(node, dict):
        return True
    if node.get("input_hash") == bedrock.input_hash(extractors.n3_payload(notes)):
        return False
    rest = notes
    for r in node.get("value") or []:
        quote = r.get("quote") if isinstance(r, dict) else None
        if isinstance(quote, str) and quote:
            rest = rest.replace(quote, " ")
    return bool(re.sub(r"[\W_]+", "", rest))

def _values(old_field: dict | None, out: dict, context: dict) -> tuple[dict | None, GateStatus]:
    node = "values_parser"
    rules = out.get("rules")
    if not isinstance(rules, list):
        return _kept(old_field, "rejected:schema")
    notes = context["notes"]

    # 2. evidence: every quote is an exact substring of the user's notes (no case or whitespace folding)
    for r in rules:
        quote = r.get("quote") if isinstance(r, dict) else None
        if not isinstance(quote, str) or not quote or quote not in notes:
            return _kept(old_field, "rejected:evidence")

    # 3. enum / range
    if not all(r.get("rule_kind") in RULE_KINDS and isinstance(r.get("params"), dict)
               and r.get("applies_to") in APPLIES_TO for r in rules):
        return _kept(old_field, "rejected:schema")
    # SPEC §3.2 / §6.2 N3: the tag and theme vocabularies and the max_pct range. A rule that fails rejects the output
    # (never coerced to a vocabulary value; the previous value stays, §1 inv. 4).
    tags = tuple(context.get("activity_tags") or ACTIVITY_TAG_VOCABULARY)
    themes = tuple(context["themes"]) if context.get("themes") is not None else theme_vocabulary()
    if not all(_n3_params_ok(r, tags, themes) for r in rules):
        return _kept(old_field, "rejected:schema")

    # Parsed rules start unapproved (SPEC §3.3 #4, §6.2 N3). A rule identical to one the user already approved
    # keeps that approval (re-parsing must not silently drop a user action).
    def ident(r: dict) -> tuple:
        return (r["quote"], r["rule_kind"], repr(sorted(r["params"].items())), r["applies_to"])

    approved = {ident(r) for r in (old_field or {}).get("value", []) if isinstance(r, dict) and r.get("approved_by_user")}

    if not _is_num(conf) or not 0 <= conf <= 1:

def _n3_params_ok(rule: dict, tags: tuple[str, ...], themes: tuple[str, ...]) -> bool:
    params = rule["params"]
    if rule["rule_kind"] == "exclude_activity":
        return isinstance(params.get("tag"), str) and params["tag"] in tags
    if rule["rule_kind"] == "max_theme_weight":
        if not (isinstance(params.get("theme"), str) and params["theme"] in themes):

    return True

def apply(weights_pct: dict[str, float], intent: dict, value_cad: float) -> tuple[dict[str, float], float]:
    """Return (new weights_pct summing to 100, new value_cad).

    Only ``cash_contribution`` changes the portfolio value. The target ticker lands exactly on
    ``target_weight_pct`` except for an under-funded cash contribution (it then gets all the cash).
    Raises ValueError if the intent cannot be applied without negative weights.
    """
    if intent.get("action", "buy") != "buy":
        raise ValueError(f"unsupported action: {intent.get('action')!r}")
    ticker = intent["ticker"]
    target = float(intent["target_weight_pct"])
    funding = intent["funding"]
    kind = funding.get("kind")
    if kind not in FUNDING_KINDS:
        raise ValueError(f"unknown funding kind: {kind!r}")
    if not 0 <= target <= 100:
        raise ValueError(f"target weight out of range: {target}")

    new = {t: float(v) for t, v in weights_pct.items()}
    cur = new.get(ticker, 0.0)
    delta = target - cur
    if delta < -_TOL:
        raise ValueError(f"target {target}% is below the current weight {cur}% of {ticker}: not a buy")
    if abs(delta) <= _TOL and kind != "cash_contribution":
        return _clean(new), value_cad
    delta = max(delta, 0.0)

    if kind == "specific":
        src = funding["ticker"]
        if src == ticker:
            raise ValueError("a ticker cannot fund itself")
        have = new.get(src, 0.0)
        if delta > have + _TOL:
            raise ValueError(f"{src} holds {have}% but {delta}% is needed")
        new[src] = max(have - delta, 0.0)
        new[ticker] = target
        return _clean(new), value_cad

    if kind == "pro_rata_all":
        if cur >= 100 - _TOL:
            raise ValueError("nothing to fund from")
        factor = (100 - target) / (100 - cur)
        out = {t: (v if t == ticker else v * factor) for t, v in new.items()}
        out[ticker] = target
        return _clean(out), value_cad

    if kind == "pro_rata_class":
        cls = funding["asset_class"]
        pool = [t for t in new if t != ticker and new[t] > 0 and _asset_class(t) == cls]
        _scale_pool(new, pool, delta)
        new[ticker] = target
        return _clean(new), value_cad

    # cash_contribution: nothing is sold; existing holdings keep their CAD value.
    amount = float(funding["amount_cad"])
    if amount < 0:
        raise ValueError("cash contribution must be >= 0")
    new_value = value_cad + amount
    cad = {t: v * value_cad / 100 for t, v in new.items()}
    need = max(target / 100 * new_value - cad.get(ticker, 0.0), 0.0)
    buy = min(amount, need)
    excess = amount - buy                    # cash beyond the target: spread pro rata over the other holdings
    others = [t for t in cad if t != ticker and cad[t] > 0]
    base = math.fsum(cad[t] for t in others)
    if excess > 1e-9 and base <= 0:
        raise ValueError("cash exceeds the target and there is no other holding to place it in")
    for t in others:
        cad[t] += excess * cad[t] / base if excess > 1e-9 else 0.0
    cad[ticker] = cad.get(ticker, 0.0) + buy
    out = {t: v / new_value * 100 for t, v in cad.items()}
    return _clean(out), new_value

            as_of: str | None = None, return_paths: bool = False,
            meta_dir: str | Path | None = None) -> dict:
    """monthly_contribution (CAD, >= 0) is added to the NAV at the end of each simulated month. `goals` need the
    reference date `as_of` (D1: portfolio.as_of); the engine never reads the wall clock."""
    c = float(monthly_contribution or 0.0)
    if not c >= 0.0:
        raise ValueError(f"monthly_contribution must be >= 0, got {monthly_contribution!r}")
    if goals and as_of is None:
        raise ValueError("goals need `as_of` (the reference date, portfolio.as_of; SPEC D1)")

    mk = market
    w = pd.Series(weights_pct, dtype=float)
    w /= w.sum()
    R = mk.r_cad[w.index]
    cov = R.cov() * 12
    var = float(w @ cov @ w)
    vol = var ** 0.5
    rc = w * (cov @ w) / var  # share of variance, sums to 1
    port = (R * w).sum(axis=1)
    if drift == "assumed":

        Rb = R - R.mean() + mu

        Rb = R
    port_b = (Rb * w).sum(axis=1)
    # local-currency vol (no FX) to isolate the currency effect
    port_loc = (mk.r_loc[w.index] * w).sum(axis=1)
    vol_loc = float(port_loc.std() * 12 ** 0.5)

    nav = (1 + port).cumprod()

    mdd_date = dd.idxmin().strftime("%Y-%m")

    rng = np.random.default_rng(seed)

def violations_added(current: list, proposed: list, ids: tuple[str, ...] | None = None) -> list[str]:
    """Ids of the hard violations the TRADE adds: violated on the proposed side and either not violated on the current
    side, or violated there and made worse by the trade (SPEC §8: "the first of the user's limits the trade crosses").

    # SPEC-AMBIGUITY: §8 does not say whether pushing further past a limit that is already breached counts as
    # crossing it. Stricter reading taken: a worsened pre-existing breach counts; an unchanged or improved one does not.
    """
    before = {_key(c): _metric(c) for c in current if c["status"] == "violation"}
    out: list[str] = []
    for c in proposed:
        if c["status"] != "violation" or (ids is not None and c["id"] not in ids) or c["id"] in out:
            continue
        k = _key(c)
        if k in before:
            b, a = before[k], _metric(c)
            if b is None or a is None:
                continue
            worse = a < b - _TOL if c["id"] in _LOWER_IS_WORSE else a > b + _TOL
            if not worse:
                continue
        out.append(c["id"])
    return out

def _grid(w0: float, target: float, step: float) -> list[float]:
    """Current weight, then the multiples of `step` above it, then the target itself."""

def _solve_mode(ev: _Evaluator, grid: list[float]) -> dict:
    """Largest grid weight that passes every check; returns {idx, binding, metrics}."""
    # 1. cheap scan: grid[0] is the current portfolio (no trade yet, nothing crossed); stop at the first failing step
    cheap_idx, stop_ids = 0, []
    for i in range(1, len(grid)):
        ids = ev.cheap(grid[i])
        if ids:
            stop_ids = ids
            break
        cheap_idx = i

    # 2. engine checks at the last cheap-passing weight with n=1000, binary search downward if they fail
    def ok_cheap_n(i: int) -> bool:
        return i == 0 or not ev.engine(grid[i], config.ENGINE_N_CHEAP)[0]

    idx = cheap_idx
    if not ok_cheap_n(idx):
        lo, hi = -1, cheap_idx - 1          # largest passing index in [lo, hi]; -1 = none passes
        while lo < hi:
            mid = (lo + hi + 1) // 2
            if ok_cheap_n(mid):
                lo = mid
            else:
                hi = mid - 1
        idx = max(lo, 0)

    # 3. confirm with the full n: step down while it fails, step up while the next grid point still passes
    while True:
        ids, m = ([], None) if idx == 0 else ev.engine(grid[idx], config.ENGINE_N)
        if ids:
            idx -= 1
            continue
        if idx == len(grid) - 1:
            binding: list[str] = []                 # the whole requested weight fits
        elif idx == cheap_idx:
            binding = stop_ids
        else:
            nids, _ = ev.engine(grid[idx + 1], config.ENGINE_N)
            if not nids:
                idx += 1
                continue
            binding = nids
        return {"idx": idx, "binding": binding, "fits": idx > 0,
                "metrics": {k: m[k] for k in _METRIC_KEYS} if m else None}

def narrate_section(facts: dict) -> dict:
    """One section: N5 -> verify -> pass `llm`; else N5 again with `retry_failures` -> pass `llm_retry`; else the
    template. Never more than 2 LLM calls. `pass` sections never reach the writer. Output that failed verification is
    never returned. A missing recording (mock) or an output that stays schema-invalid ends in the template."""
    attempts: list[dict] = []
    if facts["status"] == "pass":
        return _template_narrative(facts, attempts)
    retry_failures: list[str] | None = None
    for attempt in (1, 2):
        try:
            result = extractors.explainer_call(facts, retry_failures)
        except bedrock.LLMCacheMiss as exc:                   # nothing recorded for this input: no point in retrying
            return _template_narrative(facts, attempts, f"LLMCacheMiss: {exc}")
        except bedrock.GuardrailIntervened as exc:           # blocked, not malformed: no retry (SPEC §6.1, §9.3)
            attempts.append({"failures": [f"guardrail: {exc}"], "call_log": exc.call_log})
            return _template_narrative(facts, attempts, f"GuardrailIntervened: {exc}")
        except bedrock.LLMOutputInvalid as exc:
            failures = [f"V5: the output did not match the explainer schema: {exc}"]
            attempts.append({"failures": failures, "call_log": None})
        else:
            output = result["output"]
            failures = verify_section(output, facts)
            attempts.append({"failures": failures, "call_log": result["call_log"]})
            if not failures:
                segments = copy.deepcopy(output["segments"])
                return _narrative("llm" if attempt == 1 else "llm_retry", segments, facts, attempts)
        retry_failures = failures
    return _template_narrative(facts, attempts)

    (`V1`..`V5`); all rules are evaluated, so a bad answer lists every rule it breaks."""
    check_ids = {c["id"] for c in facts.get("checks", [])}
    event_ids = {e["event_id"] for e in facts.get("evidence_events", []) if isinstance(e, dict) and "event_id" in e}
    raw = output.get("segments") if isinstance(output, dict) else None
    segments = raw if isinstance(raw, list) else []
    v1: list[str] = []
    v2: list[str] = []
    v4: list[str] = []
    v5: list[str] = []
    cited: set[str] = set()

    if not isinstance(raw, list) or not 1 <= len(raw) <= MAX_SEGMENTS:
        v5.append(f"V5: {len(segments)} segment(s), expected 1 to {MAX_SEGMENTS}")
    allow_inference = _allow_inference(facts)
    for i, seg in enumerate(segments, 1):
        if not isinstance(seg, dict):
            v5.append(f"V5: segment {i} is not an object")
            continue
        text, kind, refs = seg.get("text"), seg.get("kind"), seg.get("refs")
        if kind not in KINDS:
            v5.append(f"V5: segment {i} has kind {kind!r}, expected one of {', '.join(KINDS)}")

            v5.append(f"V5: segment {i} is an inference but the section has no llm-derived field or evidence event")

            v5.append(f"V5: segment {i} has no text")

def template_section(facts: dict) -> dict:
    """Deterministic section text from its facts (§20.5): same facts -> byte-identical output, every number is copied
    from `facts` (so numcheck passes), every violation/warning check id is referenced (V3), at most 4 segments of
    at most 300 characters (V5), no advisory wording (V4), `inference` only where V5 allows it."""

    entries = facts["checks"]
    allow_inference = _allow_inference(facts)
    vw = [e for e in entries if e["status"] in ("violation", "warning")]
    info = [e for e in entries if e["status"] == "info"]
    unk = [e for e in entries if e["status"] == "unknown"]

    def kind_of(e: dict) -> str:
        return "inference" if e["id"] in _INFERENCE_CHECKS and allow_inference else "fact"

    segments: list[dict] = []
    room = MAX_SEGMENTS - (1 if unk else 0)
    if vw:
        slots = min(len(vw), room)
        lines = [(_line(e, loc), [e["id"]], kind_of(e)) for e in vw]
        for i in range(slots):
            chunk = lines[i:i + 1] if i < slots - 1 else lines[i:]
            text, refs = _merge([(t, r) for t, r, _ in chunk])
            kinds = {k for _, _, k in chunk}
            segments.append({"kind": "inference" if kinds == {"inference"} else "fact", "text": text, "refs": refs})
        room -= slots
        why = _why_segment(facts, vw, loc) if room > 0 else None
        if why:
            segments.append({"kind": "fact", "text": why[0], "refs": why[1]})
            room -= 1
    if info and room > 0:
        text, refs = _merge([(_line(e, loc), [e["id"]]) for e in info])
        segments.append({"kind": "fact", "text": text, "refs": refs})
    if unk:
        text, refs = _merge([(_line(e, loc), [e["id"]]) for e in unk])
        segments.append({"kind": "unknown", "text": text, "refs": refs})
    if not segments:
        segments.append({"kind": "fact", "text": _NONE[loc], "refs": ["status"]})
    return {"mode": "template", "segments": segments}

def bond_impact(duration: float, shock_bp: float, convexity: float | None = None) -> float:
    """Price change (fraction) of a bond position for a yield change of `shock_bp` basis points."""
    dy = float(shock_bp) / 10_000.0

gate(node, old_field, llm_output, context) → (new_field, status). The steps, in order:

  1. Abstain: the model said "can't tell" → keep the old value (kept:abstain).
  2. Evidence: event_ids must exist; quotes must be exact substrings → else rejected:evidence.
  3. Enum / range, checked again → else rejected:schema.
  4. Movement clamp: behaviour ±1 level, knowledge ±5 points → clamped.
  5. Confidence is stored; warn_enabled = confidence ≥ 0.5.
  6. Stamp: by: llm:<node>, prompt version, model, time, input hash.

A rejected answer always returns the previous value, so a bad answer never wipes data. N5 uses the verifier instead.
